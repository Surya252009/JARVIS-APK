OLLAMA_URL = "http://localhost:11434/api/generate"
OLLAMA_MODEL = "llama3.2"
WAKE_WORD = "jarvis"

SYSTEM_PROMPT = (
    "You are J.A.R.V.I.S., Tony Stark's AI assistant. "
    "Keep answers concise (1-2 sentences), witty, British, and helpful. "
    "Address the user as 'Sir' or 'Boss'."

engine = pyttsx3.init()
engine.setProperty("rate", 185)

def speak(text: str):
    print(f"\n[J.A.R.V.I.S.]: {text}")
    engine.say(text)
    engine.runAndWait()

def execute_local_skill(command: str):
    cmd = command.lower().strip()
    if "time" in cmd:
        now = datetime.datetime.now().strftime("%I:%M %p")
        return f"The current time is {now}, Sir."
    if "system status" in cmd or "diagnostics" in cmd:
        cpu = psutil.cpu_percent(interval=0.5)
        mem = psutil.virtual_memory().percent
        return f"All systems nominal. CPU load is at {cpu} percent and memory allocation is at {mem} percent."
    if cmd.startswith("open "):
        target = cmd.replace("open ", "").strip()
        if "youtube" in target:
            webbrowser.open("https://www.youtube.com")
            return "Opening YouTube in your primary browser, Sir."
        if "google" in target:
            webbrowser.open("https://www.google.com")
            return "Launching Google search interface."
        if "calculator" in target:
            subprocess.Popen("calc.exe" if os.name == "nt" else "gnome-calculator", shell=True)
            return "Launching system calculator."
    return None

def ask_ollama(prompt: str) -> str:
    try:
        resp = requests.post(
            OLLAMA_URL,
            json={
                "model": OLLAMA_MODEL,
                "prompt": f"{SYSTEM_PROMPT}\nUser: {prompt}\nJARVIS:",
                "stream": False,
            },
            timeout=15,
        )
        if resp.status_code == 200:
            return resp.json().get("response", "").strip()
    except Exception:
        pass
    return "My connection to the local Ollama neural server is offline, Sir, but all core protocols remain operational."

def listen_command() -> str:
    recognizer = sr.Recognizer()
    with sr.Microphone() as source:
        print("\n[Listening for command...]")
        recognizer.adjust_for_ambient_noise(source, duration=0.4)
        try:
            audio = recognizer.listen(source, timeout=5, phrase_time_limit=8)
            text = recognizer.recognize_google(audio)
            print(f"[You]: {text}")
            return text
        except Exception:
            return ""

def main():
    speak("J.A.R.V.I.S. Mark 7 online. All systems nominal. How may I assist you today, Sir?")
    while True:
        try:
            user_input = input("\nPress ENTER to speak, or type a command ('exit' to quit): ").strip()
            if user_input.lower() in ("exit", "quit", "shutdown"):
                speak("Powering down systems. Goodbye, Sir.")
                break
            query = user_input if user_input else listen_command()
            if not query:
                continue
            skill_reply = execute_local_skill(query)
            if skill_reply:
                speak(skill_reply)
            else:
                speak(ask_ollama(query))
        except KeyboardInterrupt:
            speak("Shutting down. Goodbye, Sir.")
            break

if __name__ == "__main__":
    main()

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type, X-JARVIS-KEY",

function timingSafeEqualStr(a, b) {
  const strA = String(a || "");
  const strB = String(b || "");
  if (strA.length !== strB.length || strA.length === 0) return false;
  let diff = 0;
  for (let i = 0; i < strA.length; i++) {
    diff |= strA.charCodeAt(i) ^ strB.charCodeAt(i);
  }
  return diff === 0;
}

export default {
  async fetch(request, env) {
    if (request.method === "OPTIONS") {
      return new Response(null, { status: 204, headers: corsHeaders });
    }

    const url = new URL(request.url);

    // 1. Safe Diagnostics / Health Check Endpoint (Never exposes secret values)
    if (request.method === "GET" && (url.pathname === "/" || url.pathname === "/health")) {
      return Response.json(
        {
          ok: true,
          service: "jarvis-voicemonkey-bridge",
          triggers: DEVICES,
          tokenConfigured: Boolean(env.VOICEMONKEY_TOKEN),
          bridgeKeyConfigured: Boolean(env.JARVIS_BRIDGE_KEY)
        },
        { status: 200, headers: corsHeaders }
      );
    }

    // 2. Trigger Endpoint (POST /trigger)
    if (request.method !== "POST") {
      return Response.json(
        { success: false, error: "Use POST /trigger or GET /health" },
        { status: 405, headers: corsHeaders }
      );
    }

    if (url.pathname !== "/trigger") {
      return Response.json(
        { success: false, error: "Endpoint not found. Use POST /trigger" },
        { status: 404, headers: corsHeaders }
      );
    }

    if (!env.JARVIS_BRIDGE_KEY) {
      return Response.json(
        { success: false, error: "Server secret JARVIS_BRIDGE_KEY is not configured in Cloudflare Worker" },
        { status: 500, headers: corsHeaders }
      );
    }

    const suppliedKey = request.headers.get("X-JARVIS-KEY") || "";
    if (!timingSafeEqualStr(suppliedKey, env.JARVIS_BRIDGE_KEY)) {
      return Response.json(
        { success: false, error: "Unauthorized bridge key (X-JARVIS-KEY mismatch)" },
        { status: 401, headers: corsHeaders }
      );
    }

    if (!env.VOICEMONKEY_TOKEN) {
      return Response.json(
        { success: false, error: "Server secret VOICEMONKEY_TOKEN is not configured in Cloudflare Worker" },
        { status: 500, headers: corsHeaders }
      );
    }

    let body;
    try {
      body = await request.json();
    } catch {
      return Response.json(
        { success: false, error: "Invalid JSON body" },
        { status: 400, headers: corsHeaders }
      );
    }

    const action = String(body.action || "").toLowerCase().trim();
    const device = DEVICES[action];
    if (!device) {
      return Response.json(
        { success: false, error: "Action must be 'on' or 'off'" },
        { status: 400, headers: corsHeaders }
      );
    }

    try {
      const apiResponse = await fetch("https://api-v3.voicemonkey.io/trigger", {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${env.VOICEMONKEY_TOKEN}`,
          "Content-Type": "application/json; charset=utf-8"
        },
        body: JSON.stringify({
          token: env.VOICEMONKEY_TOKEN,
          device
        })
      });

      const result = await apiResponse.json().catch(() => ({}));
      if (!apiResponse.ok || result.success !== true) {
        return Response.json(
          {
            success: false,
            action,
            device,
            error: result.message || result.error || `Voice Monkey API returned HTTP ${apiResponse.status}`
          },
          { status: 502, headers: corsHeaders }
        );
      }

      return Response.json(
        {
          success: true,
          action,
          device,
          upstream: result.data || "OK",
          timestamp: new Date().toISOString()
        },
        { status: 200, headers: corsHeaders }
      );
    } catch {
      return Response.json(
        { success: false, action, device, error: "Could not reach Voice Monkey V3 API" },
        { status: 502, headers: corsHeaders }
      );
    }
  }
